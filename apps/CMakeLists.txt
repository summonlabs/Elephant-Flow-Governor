add_library(efg_cli_support STATIC cli.cpp)
target_link_libraries(efg_cli_support PUBLIC efg::efg)
target_include_directories(efg_cli_support PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(efg_cli_support PUBLIC cxx_std_20)
if(EFG_WARNINGS_AS_ERRORS)
    efg_apply_warnings(efg_cli_support)
endif()
if(EFG_SANITIZE)
    efg_apply_sanitizers(efg_cli_support)
endif()
set_target_properties(efg_cli_support PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)

foreach(tool governor coordinator worker bench)
    add_executable(efg-${tool} efg_${tool}.cpp)
    target_link_libraries(efg-${tool} PRIVATE efg_cli_support)
    set_target_properties(efg-${tool} PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
    if(EFG_WARNINGS_AS_ERRORS)
        efg_apply_warnings(efg-${tool})
    endif()
    if(EFG_SANITIZE)
        efg_apply_sanitizers(efg-${tool})
    endif()
endforeach()
